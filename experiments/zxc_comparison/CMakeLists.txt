cmake_minimum_required(VERSION 3.21)
project(cng_zxc_comparison LANGUAGES C CXX)
set(ZXC_NATIVE_ARCH OFF CACHE BOOL "" FORCE)
set(ZXC_ENABLE_LTO OFF CACHE BOOL "" FORCE)
set(ZXC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZXC_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(ZXC_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../build/zxc-study/source zxc)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
add_executable(compare probe.cpp ../whole_codec/match_bounded.cpp ../decode_tables/huffman_batch.cpp)
target_compile_features(compare PRIVATE cxx_std_20)
set_target_properties(compare PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(compare PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(compare PRIVATE CNG_SHORT_STREAM=1 CNG_UNIFORM_PROOF=1 CNG_SPLIT_HISTOGRAM=1)
target_include_directories(compare PRIVATE ../entropy "${ZSTD_INCLUDE_DIR}")
target_link_libraries(compare PRIVATE zxc::zxc_lib "${ZSTD_LIBRARY}")
add_executable(huffman huffman.cpp huffman_bridge.c ../decode_tables/huffman_batch.cpp)
target_compile_features(huffman PRIVATE cxx_std_20)
set_target_properties(huffman PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(huffman PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(huffman PRIVATE CNG_SHORT_STREAM=1 CNG_UNIFORM_PROOF=1 CNG_SPLIT_HISTOGRAM=1)
target_include_directories(huffman PRIVATE ../entropy ../../build/zxc-study/source/src/lib ../../build/zxc-study/source/src/lib/vendors)
target_link_libraries(huffman PRIVATE zxc::zxc_lib)
