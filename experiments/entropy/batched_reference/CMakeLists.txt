cmake_minimum_required(VERSION 3.20)
project(cng_entropy_trial LANGUAGES CXX)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(cng_entropy_probe probe.cpp rans.cpp)
target_compile_features(cng_entropy_probe PRIVATE cxx_std_20)
set_target_properties(cng_entropy_probe PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(cng_entropy_probe PRIVATE "${ZSTD_INCLUDE_DIR}")
target_link_libraries(cng_entropy_probe PRIVATE "${ZSTD_LIBRARY}" "${ZLIB_LIBRARIES}")
if(NOT APPLE)
    target_include_directories(cng_entropy_probe PRIVATE "${ZLIB_INCLUDE_DIRS}")
endif()
if(MSVC)
    target_compile_options(cng_entropy_probe PRIVATE /W4 /WX)
else()
    target_compile_options(cng_entropy_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
