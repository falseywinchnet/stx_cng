cmake_minimum_required(VERSION 3.20)
project(cng_entropy_reduction LANGUAGES CXX)
find_package(ZLIB REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
add_library(reduction_shared STATIC ../../src/prediction_parse.cpp ../../src/prediction_restore.cpp ../../src/raw_restore.cpp ../../src/direct_words.cpp)
add_library(reduction_reference STATIC ../../src/codec.cpp)
target_link_libraries(reduction_reference PRIVATE reduction_shared "${ZSTD_LIBRARY}" "${ZLIB_LIBRARIES}")
set(RESEARCH_TARGETS reduction_shared reduction_reference)
foreach(VARIANT uniform histogram combined)
    add_library(candidate_${VARIANT} STATIC ../whole_codec/candidate_codec.cpp ../whole_codec/match_bounded.cpp huffman_reduced.cpp)
    target_compile_definitions(candidate_${VARIANT} PRIVATE CNG_SHORT_STREAM=1)
    if(NOT VARIANT STREQUAL "histogram")
        target_compile_definitions(candidate_${VARIANT} PRIVATE CNG_UNIFORM_PROOF=1)
    endif()
    if(NOT VARIANT STREQUAL "uniform")
        target_compile_definitions(candidate_${VARIANT} PRIVATE CNG_SPLIT_HISTOGRAM=1)
    endif()
    target_link_libraries(candidate_${VARIANT} PRIVATE reduction_shared "${ZLIB_LIBRARIES}")
    add_executable(cng_reduced_${VARIANT} ../whole_codec/probe.cpp)
    target_link_libraries(cng_reduced_${VARIANT} PRIVATE reduction_reference candidate_${VARIANT})
    add_executable(cng_proof_${VARIANT} proof_checks.cpp)
    target_compile_definitions(cng_proof_${VARIANT} PRIVATE CNG_SHORT_STREAM=1)
    target_link_libraries(cng_proof_${VARIANT} PRIVATE candidate_${VARIANT})
    list(APPEND RESEARCH_TARGETS candidate_${VARIANT} cng_reduced_${VARIANT} cng_proof_${VARIANT})
endforeach()
foreach(TARGET ${RESEARCH_TARGETS})
    target_compile_features(${TARGET} PUBLIC cxx_std_20)
    set_target_properties(${TARGET} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${TARGET} PRIVATE ../../include ../../src ../entropy ../whole_codec "${ZSTD_INCLUDE_DIR}")
    if(NOT APPLE)
        target_include_directories(${TARGET} PRIVATE "${ZLIB_INCLUDE_DIRS}")
    endif()
    if(MSVC)
        target_compile_options(${TARGET} PRIVATE /W4 /WX)
    else()
        target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
