cmake_minimum_required(VERSION 3.20)
project(cng_whole_trial LANGUAGES CXX)
find_package(ZLIB REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
add_library(whole_shared STATIC ../../src/prediction_parse.cpp ../../src/prediction_restore.cpp ../../src/raw_restore.cpp ../../src/direct_words.cpp)
add_library(whole_reference STATIC ../../src/codec.cpp)
add_library(whole_candidate STATIC candidate_codec.cpp match_bounded.cpp ../entropy/huffman_tail_split.cpp)
add_executable(cng_whole_probe probe.cpp)
foreach(TARGET whole_shared whole_reference whole_candidate cng_whole_probe)
    target_compile_features(${TARGET} PUBLIC cxx_std_20)
    set_target_properties(${TARGET} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${TARGET} PRIVATE ../../include ../../src ../entropy "${ZSTD_INCLUDE_DIR}")
    if(NOT APPLE)
        target_include_directories(${TARGET} PRIVATE "${ZLIB_INCLUDE_DIRS}")
    endif()
    if(MSVC)
        target_compile_options(${TARGET} PRIVATE /W4 /WX)
    else()
        target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
target_compile_definitions(whole_candidate PRIVATE CNG_SHORT_STREAM=1)
target_link_libraries(whole_reference PRIVATE whole_shared "${ZSTD_LIBRARY}" "${ZLIB_LIBRARIES}")
target_link_libraries(whole_candidate PRIVATE whole_shared "${ZLIB_LIBRARIES}")
target_link_libraries(cng_whole_probe PRIVATE whole_reference whole_candidate)
add_executable(cng_whole_bounds bounds_checks.cpp match_bounded.cpp ../entropy/huffman_tail_split.cpp)
target_compile_features(cng_whole_bounds PRIVATE cxx_std_20)
set_target_properties(cng_whole_bounds PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(cng_whole_bounds PRIVATE ../entropy)
target_compile_definitions(cng_whole_bounds PRIVATE CNG_SHORT_STREAM=1)
if(MSVC)
    target_compile_options(cng_whole_bounds PRIVATE /W4 /WX)
else()
    target_compile_options(cng_whole_bounds PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
