cmake_minimum_required(VERSION 3.21)
project(cngcompress_integration LANGUAGES C CXX)
add_subdirectory(../.. cng)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
find_package(ZLIB REQUIRED)
add_library(release_v2 STATIC ../release_v2/src/codec.cpp ../release_v2/src/collection.cpp)
target_compile_features(release_v2 PRIVATE cxx_std_20)
target_compile_definitions(release_v2 PRIVATE stx_cng=cng_release_v2 stx_cng_collection=cng_release_v2_collection)
target_include_directories(release_v2 PRIVATE ../release_v2/include "${ZSTD_INCLUDE_DIR}")
target_link_libraries(release_v2 PRIVATE stx_cng "${ZSTD_LIBRARY}" "${ZLIB_LIBRARIES}")
add_executable(probe probe.cpp)
add_executable(collections collections.cpp)
foreach(target probe collections)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_link_libraries(${target} PRIVATE stx_cng release_v2)
endforeach()
foreach(target release_v2 probe collections)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
