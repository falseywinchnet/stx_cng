cmake_minimum_required(VERSION 3.20)
project(cng_raw_fallback LANGUAGES CXX)
find_package(ZLIB REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
add_library(fallback_shared STATIC ../../src/prediction_parse.cpp ../../src/prediction_restore.cpp ../../src/raw_restore.cpp ../../src/direct_words.cpp)
add_library(fallback_reference STATIC ../../src/codec.cpp)
add_library(fallback_candidate STATIC candidate_codec.cpp match_limited.cpp huffman_limited.cpp)
target_link_libraries(fallback_reference PRIVATE fallback_shared "${ZSTD_LIBRARY}" "${ZLIB_LIBRARIES}")
target_link_libraries(fallback_candidate PRIVATE fallback_shared "${ZLIB_LIBRARIES}")
target_compile_definitions(fallback_candidate PRIVATE CNG_SHORT_STREAM=1 CNG_SPLIT_HISTOGRAM=1 CNG_UNIFORM_PROOF=1)
add_executable(cng_fallback_probe ../whole_codec/probe.cpp)
target_link_libraries(cng_fallback_probe PRIVATE fallback_reference fallback_candidate)
add_executable(cng_fallback_checks checks.cpp)
target_link_libraries(cng_fallback_checks PRIVATE fallback_candidate)
add_executable(cng_fallback_huffman huffman_checks.cpp)
target_link_libraries(cng_fallback_huffman PRIVATE fallback_candidate)
target_compile_definitions(cng_fallback_huffman PRIVATE CNG_SHORT_STREAM=1)
foreach(TARGET fallback_shared fallback_reference fallback_candidate cng_fallback_probe cng_fallback_checks cng_fallback_huffman)
    target_compile_features(${TARGET} PUBLIC cxx_std_20)
    set_target_properties(${TARGET} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${TARGET} PRIVATE ../../include ../../src ../entropy ../whole_codec "${ZSTD_INCLUDE_DIR}")
    if(NOT APPLE)
        target_include_directories(${TARGET} PRIVATE "${ZLIB_INCLUDE_DIRS}")
    endif()
    if(MSVC)
        target_compile_options(${TARGET} PRIVATE /W4 /WX)
    else()
        target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
