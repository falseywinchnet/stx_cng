cmake_minimum_required(VERSION 3.20)
project(cngcompress_gate LANGUAGES CXX)
add_subdirectory(../decode_tables decode)
option(CNG_GATE_SHORT_TABLE "Isolate short-stream table construction" OFF)
if(CNG_GATE_SHORT_TABLE)
    get_target_property(CNG_GATE_SOURCES whole_candidate SOURCES)
    list(REMOVE_ITEM CNG_GATE_SOURCES huffman_batch.cpp)
    list(APPEND CNG_GATE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/huffman_short.cpp")
    set_property(TARGET whole_candidate PROPERTY SOURCES "${CNG_GATE_SOURCES}")
endif()
find_package(ZLIB REQUIRED)
add_library(reference_collection STATIC ../../src/collection.cpp)
add_library(candidate_collection STATIC candidate_collection.cpp)
add_executable(collections collections.cpp)
foreach(TARGET reference_collection candidate_collection collections)
    target_compile_features(${TARGET} PRIVATE cxx_std_20)
    set_target_properties(${TARGET} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${TARGET} PRIVATE ../../include)
    target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
target_link_libraries(reference_collection PRIVATE whole_reference "${ZLIB_LIBRARIES}")
target_link_libraries(candidate_collection PRIVATE whole_candidate "${ZLIB_LIBRARIES}")
target_link_libraries(collections PRIVATE reference_collection candidate_collection)
